function syncSelection(){

  const count =
    totalCookies();


  cards.forEach(card => {

    const item =
      selected.find(
        item =>
          item.name ===
          card.dataset.cookie
      );


    card.classList.toggle(
      'selected',
      Boolean(item)
    );


    /*
      Se já chegamos a 3 cookies,
      os sabores ainda não selecionados
      ficam visualmente bloqueados.
    */

    card.classList.toggle(
      'limit-reached',

      count >= MAX_COOKIES &&
      !item
    );

  });


  selectionCountEl.textContent =
    `${count}/${MAX_COOKIES}`;


  selectionCountEl.classList.toggle(
    'full',
    count === MAX_COOKIES
  );


  /*
    Só permite continuar
    se existir pelo menos 1 cookie.
  */

  continueButton.disabled =
    count === 0;

}


/*
  Clique nos sabores.
*/

cards.forEach(card => {

  card.addEventListener(
    'click',
    () => {

      const name =
        card.dataset.cookie;


      const found =
        selected.findIndex(
          item =>
            item.name === name
        );


      /*
        Se clicar novamente em um sabor
        já selecionado, remove o sabor.
      */

      if(found >= 0){

        selected.splice(
          found,
          1
        );

      }

      /*
        Se ainda houver espaço no pedido,
        adiciona 1 unidade.
      */

      else if(
        totalCookies() <
        MAX_COOKIES
      ){

        selected.push({

          name:name,

          price:Number(
            card.dataset.price
          ),

          qty:1

        });

      }


      syncSelection();

    }
  );

});


/* =========================================================
   TELA DE CONFIRMAÇÃO
========================================================= */

function renderOrder(){

  const count =
    totalCookies();


  const total =
    selected.reduce(

      (sum,item) =>
        sum +
        item.price *
        item.qty,

      0

    );


  orderItems.innerHTML =
    selected.map(
      (item,index) => {

        /*
          Se o total já for 3,
          botão + fica bloqueado.
        */

        const plusDisabled =
          count >= MAX_COOKIES
          ? 'disabled'
          : '';


        return `

          <div class="order-item">

            <span
              class="cookie ${cookieClass(item.name)}"
            >
              ${chips()}
            </span>


            <div class="order-item-info">

              <strong>
                ${item.name}
              </strong>

              <span>
                ${money(item.price)} cada
              </span>

            </div>


            <div class="qty-control">

              <button
                type="button"
                class="qty-button"
                data-action="minus"
                data-index="${index}"
              >
                −
              </button>


              <span class="qty-count">
                ${item.qty}
              </span>


              <button
                type="button"
                class="qty-button plus"
                data-action="plus"
                data-index="${index}"
                ${plusDisabled}
              >
                +
              </button>

            </div>

          </div>

        `;

      }

    ).join('');


  subtotalEl.textContent =
    money(total);

  totalEl.textContent =
    money(total);

  finalTotalEl.textContent =
    money(total);


  /*
    Botões + e - das quantidades.
  */

  orderItems
    .querySelectorAll(
      '.qty-button'
    )
    .forEach(
      button => {

        button.addEventListener(
          'click',
          () => {

            const index =
              Number(
                button.dataset.index
              );


            const item =
              selected[index];


            if(!item){
              return;
            }


            /*
              BOTÃO +
            */

            if(
              button.dataset.action ===
              'plus'
            ){

              /*
                IMPORTANTE:

                máximo de 3 cookies
                no pedido INTEIRO.
              */

              if(
                totalCookies() <
                MAX_COOKIES
              ){

                item.qty += 1;

              }

            }


            /*
              BOTÃO -
            */

            else if(
              item.qty > 1
            ){

              item.qty -= 1;

            }


            syncSelection();

            renderOrder();

          }
        );

      }

    );

}

/* =========================================================

========================================================= */

function setProgress(
  percent,
  stage,
  delivered = 0,
  total = totalCookies()
){

  document
    .getElementById(
      'progress-percent'
    )
    .textContent =
      `${percent}%`;


  document
    .getElementById(
      'progress-fill'
    )
    .style.width =
      `${percent}%`;


  const first =
    selected[0];


  if(first){

    document
      .getElementById(
        'progress-cookie-name'
      )
      .textContent =

      selected.length === 1

      ? first.name

      : `${total} cookies`;

  }


  const s1 =
    document.getElementById(
      'status-1'
    );

  const s2 =
    document.getElementById(
      'status-2'
    );

  const s3 =
    document.getElementById(
      'status-3'
    );


  [
    s1,
    s2,
    s3
  ].forEach(
    status => {

      status.classList.remove(
        'done',
        'current',
        'pending'
      );

    }
  );


  /*
    Pedido recebido
    sempre concluído
    durante o progresso.
  */

  s1.classList.add(
    'done'
  );


  /*
    ETAPA 1:
    separando cookies
  */

  if(stage === 1){

    s2.classList.add(
      'current'
    );

    s3.classList.add(
      'pending'
    );

  }


  /*
    ETAPA 2:
    liberando na bandeja
  */

  else if(stage === 2){

    s2.classList.add(
      'done'
    );

    s3.classList.add(
      'current'
    );

  }


  /*
    ETAPA 3:
    tudo entregue
  */

  else if(stage === 3){

    s2.classList.add(
      'done'
    );

    s3.classList.add(
      'done'
    );

  }


  const releaseText =
    s3.querySelector(
      'small'
    );


  /*
    Mostra quantos cookies
    já passaram pelo sensor.
  */

  if(
    stage >= 2 &&
    total > 1
  ){

    releaseText.textContent =
      `${delivered} de ${total} cookies confirmados pelo sensor.`;

  }

  else if(
    stage === 3
  ){

    releaseText.textContent =
      'Entrega confirmada pelo sensor.';

  }

  else{

    releaseText.textContent =
      'Quase pronto!';

  }

}

    'waiting',
    'Enviando pedido ao ESP32',
    'Aguardando a máquina aceitar o pedido.'
  );


  setMachineConnection(
    'connecting',
    'Conectando ao ESP'
  );


  /*
    MODO DE SIMULAÇÃO.
  */

  if(
    CONFIG.SIMULAR_ESP32
  ){

    startMockOrder();

    return;

  }


  /*
    ESP32 REAL.
  */

  try{

    const response =
      await fetch(

        `${CONFIG.ESP32_BASE_URL}/api/pedido`,

        {

          method:'POST',

          headers:{
            'Content-Type':
              'application/json'
          },

          body:
            JSON.stringify(
              orderPayload()
            )

        }

      );


    if(!response.ok){

      throw new Error(
        `ESP32 respondeu HTTP ${response.status}`
      );

    }


    setMachineConnection(
      'online',
      'Máquina online'
    );


    setProgress(
      36,
      1,
      0,
      totalCookies()
    );


    setSensorFeedback(
      'waiting',
      'Pedido recebido pela máquina',
      'Aguardando o início da separação.'
    );


    schedulePoll(
      0
    );

  }

  catch(error){

    orderInProgress =
      false;


    confirmButton.disabled =
      false;


    setMachineConnection(
      'offline',
      'ESP desconectado'
    );


    setSensorFeedback(
      'error',
      'Falha de comunicação',
      `${error.message}. O pedido não será marcado como entregue.`
    );

  }

}


/* =========================================================
   AGENDAMENTO DE CONSULTA
========================================================= */

function schedulePoll(
  delay =
    CONFIG.POLL_INTERVAL_MS
){

  clearTimeout(
    pollTimer
  );


  pollTimer =
    setTimeout(
      readEspStatus,
      delay
    );

}


/* =========================================================
   CONSULTA STATUS DO ESP32
========================================================= */

async function readEspStatus(){

  if(
    !orderInProgress
  ){
    return;
  }


  try{

    const response =
      await fetch(

        `${CONFIG.ESP32_BASE_URL}/api/pedido/status`,

        {

          method:'GET',

          cache:'no-store'

        }

      );


    if(!response.ok){

      throw new Error(
        `HTTP ${response.status}`
      );

    }


    const data =
      await response.json();


    setMachineConnection(
      'online',
      'Máquina online'
    );


    applyEspStatus(
      data
    );


    /*
      Continua consultando
      enquanto o pedido
      estiver em andamento.
    */

    if(
      orderInProgress
    ){

      schedulePoll();

    }

  }

  catch(error){

    setMachineConnection(
      'offline',
      'Sem comunicação'
    );


    setSensorFeedback(
      'error',
      'Comunicação interrompida',
      'Tentando reconectar ao ESP32. A IHM não avançará para 100% sem confirmação.'
    );


    /*
      Tenta novamente.
    */

    schedulePoll(
      1200
    );

  }

}


/* =========================================================
   PROCESSA RESPOSTA DO ESP32
========================================================= */

function applyEspStatus(data){

  const status =
    String(
      data.status || ''
    )
    .toLowerCase();


  const sensor =
    String(
      data.sensor ||
      data.mensagem ||
      ''
    )
    .toLowerCase();


  /*
    Quantidade esperada
    segundo a IHM.
  */

  const expected =
    totalCookies();


  /*
    Quantidade já confirmada
    pelos sensores.
  */

  const delivered =
    Number(

      data.cookies_entregues
      ??
      data.entregues
      ??
      0

    );


  /*
    Total informado pelo ESP.
  */

  const reportedTotal =
    Number(

      data.total_cookies
      ??
      data.total
      ??
      expected

    );


  /*
    Verifica se o ESP informou
    explicitamente que houve
    confirmação do sensor.
  */

  const sensorSaysDelivered =

    data.cookie_entregue === true

    ||

    sensor.includes(
      'ok cookie entregue'
    )

    ||

    sensor.includes(
      'cookie entregue'
    );


  /* =========================
     ERRO
  ========================= */

  if(
    status === 'erro'
  ){

    const msg =
      data.mensagem
      ||
      'A máquina informou uma falha durante a preparação.';


    setSensorFeedback(
      'error',
      'Erro na máquina',
      msg
    );


    setProgress(

      Math.min(
        99,
        Number(
          data.progresso
        )
        ||
        36
      ),

      2,

      delivered,

      expected

    );


    return;

  }


  /* =========================
     PEDIDO RECEBIDO
  ========================= */

  if(
    status === 'recebido'
  ){

    setProgress(
      36,
      1,
      delivered,
      expected
    );


    setSensorFeedback(
      'waiting',
      'Pedido recebido',
      data.mensagem ||
      'Aguardando separação.'
    );


    return;

  }


  /* =========================
     PREPARANDO
  ========================= */

  if(
    status === 'preparando'
    ||
    status === 'separando'
  ){

    setProgress(
      50,
      1,
      delivered,
      expected
    );


    setSensorFeedback(
      'waiting',
      'Separando cookies',
      data.mensagem ||
      'Motores em operação.'
    );


    return;

  }


  /* =========================
     COOKIE(S) PASSANDO PELO SENSOR
  ========================= */

  if(

    status === 'liberando'

    ||

    sensorSaysDelivered

    ||

    delivered > 0

  ){

    /*
      Garante que não ultrapasse
      o total esperado.
    */

    const safeDelivered =
      Math.min(
        delivered,
        expected
      );


    /*
      A porcentagem pode avançar
      entre 66 e 99%.

      100% é reservado
      exclusivamente para quando
      TODOS os cookies forem
      confirmados pelo sensor.
    */

    const partialPercent =
      Math.min(

        99,

        66 +

        Math.floor(

          33 *
          (
            safeDelivered /
            expected
          )

        )

      );


    setProgress(
      partialPercent,
      2,
      safeDelivered,
      expected
    );


    if(
      safeDelivered > 0
    ){

      setSensorFeedback(

        'ok',

        'Sensor detectou entrega',

        `${safeDelivered} de ${expected} cookies confirmados.`

      );

    }

    else{

      setSensorFeedback(

        'waiting',

        'Liberando na bandeja',

        data.mensagem
        ||
        'Aguardando confirmação do sensor de passagem.'

      );

    }

  }


  /*
    =====================================
    REGRA MAIS IMPORTANTE DA IHM
    =====================================

    A tela NÃO pode chegar a 100%
    apenas porque passou algum tempo.

    Ela só finaliza quando:

    1. A quantidade entregue
       for igual ao total do pedido.

    2. O ESP32 confirmar que
       o sensor detectou a entrega.
  */


  const allDelivered =

    delivered >= expected

    &&

    reportedTotal >= expected;


  /*
    A contagem sozinha
    não é suficiente.

    Precisamos também do
    sinal de confirmação
    vindo do sensor.
  */

  if(
    allDelivered
    &&
    sensorSaysDelivered
  ){

    finishOrder(
      delivered,
      expected
    );

  }

}


/* =========================================================
   FINALIZAÇÃO DO PEDIDO
========================================================= */

function finishOrder(
  delivered,
  total
){

  orderInProgress =
    false;


  stopOrderPolling();


  /*
    AGORA SIM:
    100%
  */

  setProgress(
    100,
    3,
    delivered,
    total
  );


  setSensorFeedback(

    'ok',

    'Entrega confirmada',

    `OK cookie entregue — ${delivered} de ${total} cookies detectados.`

  );


  /*
    Pequeno atraso para o usuário
    conseguir visualizar o 100%.
  */

  setTimeout(
    () => {

      showScreen(
        'success'
      );

    },

    650

  );

}


/* =========================================================
   PARA CONSULTAS / TIMERS
========================================================= */

function stopOrderPolling(){

  clearTimeout(
    pollTimer
  );


  clearTimeout(
    mockTimer
  );


  pollTimer =
    null;


  mockTimer =
    null;

}


/* =========================================================
   SIMULAÇÃO DO ESP32
========================================================= */

/*
  Esta função existe apenas
  para testar a interface
  sem o hardware conectado.
*/

function startMockOrder(){

  setMachineConnection(
    'online',
    'ESP simulado'
  );


  setProgress(
    36,
    1,
    0,
    totalCookies()
  );


  setSensorFeedback(

    'waiting',

    'Pedido recebido',

    'Modo de simulação ativado.'

  );


  const expected =
    totalCookies();


  let delivered =
    0;


  /*
    Simula o início
    da liberação.
  */

  mockTimer =
    setTimeout(

      () => {

        if(
          !orderInProgress
        ){
          return;
        }


        setProgress(
          66,
          2,
          delivered,
          expected
        );


        setSensorFeedback(

          'waiting',

          'Liberando na bandeja',

          'Aguardando o sensor detectar os cookies.'

        );


        /*
          Simula uma detecção
          de sensor por cookie.
        */

        const confirmOne =
          () => {

            if(
              !orderInProgress
            ){
              return;
            }


            delivered += 1;


            /*
              Simula exatamente
              a resposta que
              esperamos receber
              do ESP32 real.
            */

            applyEspStatus({

              status:
                delivered >= expected
                ? 'entregue'
                : 'liberando',


              sensor:
                'ok cookie entregue',


              cookies_entregues:
                delivered,


              total_cookies:
                expected

            });


            /*
              Se ainda faltarem cookies,
              aguarda e simula
              o próximo.
            */

            if(
              delivered < expected
            ){

              mockTimer =
                setTimeout(
                  confirmOne,
                  1100
                );

            }

          };


        mockTimer =
          setTimeout(
            confirmOne,
            1100
          );

      },

      1200

    );

}
